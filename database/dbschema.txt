
Project q_cash {
  database_type: 'PostgreSQL'
  Note: '''
    Q-Cash operational database.

    MONEY CONVENTION:
    All monetary values are BIGINT in minor currency units.
    For KES, 100 minor units = KSh 1.00.
    Example: KSh 100.50 = 10050.

    RATE CONVENTION:
    Percentage rates are stored as integer basis points.
    100 basis points = 1%.
    Example: 10% = 1000 basis points.

    Authentication is managed by Markt/LaravelAuth.
    The users table below is a reference only.
    Do not create or recreate authentication tables.
  '''
}

  Indexes {

}
